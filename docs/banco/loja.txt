LOJA OFICIAL — TABELAS, RELACOES E REGRAS
Revisao: 06/11/2026 | Fontes: schema Prisma e migration 20261106020000_shop.
Este arquivo descreve a estrutura versionada. A migration nao foi aplicada ao banco local nesta implementacao.
Nenhuma credencial, documento ou registro de cliente foi exportado.

9 novas tabelas. User recebe relacoes virtuais com pedidos, pagamentos confirmados, movimentacoes e cotacoes.
GatewayIntent recebe shopOrderId UUID opcional e UNIQUE; a CHECK exige exatamente um destino entre ticketId, chargeId e shopOrderId.
O recebimento da loja usa ShopPayment; a compra com cashback dos parceiros continua em PaymentCharge.

RELACOES
User 1:N ShopOrder (comprador), ShopShippingQuote (cotacao), StockMovement (responsavel).
User 1:N ShopPayment (confirmador administrativo opcional).
Product 1:N ProductVariant. ProductVariant 1:N ShopOrderItem e StockMovement.
ShopOrder 1:N ShopOrderItem e StockMovement; ShopOrder 0..1 ShopPayment e GatewayIntent.
ShopSettings tem uma linha de configuracao da instancia (id main).
ShippingRule guarda tabela propria; ShopShippingQuote guarda opcoes cotadas e hash de carrinho.

CONVENCOES
UUIDs sao gerados pelo Prisma. Valores de moeda sao INTEGER em centavos BRL. Peso em gramas, medidas em cm.
CEPs e telefones sao normalizados; mascaras sao de apresentacao. Datas armazenam instantes, apresentacao em America/Fortaleza.
@updatedAt e mantido pelo Prisma. Campos relacionais/listas do schema sao virtuais, exceto arrays explicitamente persistidos (Product.images).

REGRAS DE ESTOQUE E IDENTIFICACAO
ProductVariant.stock = saldo fisico; disponivel = fisico menos PENDING nao expirados.
Reserva dura 30 minutos; cotacao dura 10 minutos e nao reserva estoque. EXPIRED e calculado para exibicao.
ShopOrder.number e sequencial para atendimento; id UUID e identificador interno, nao uma credencial.
ShopOrder.userId + requestKey e UNIQUE; requestHash impede repetir a mesma chave com outros dados.
Itens guardam snapshot de nome, SKU, variacao, imagem, custo e preco. shippingSnapshot preserva a opcao tecnica cotada.
Recebimento baixa estoque e grava SALE/recibo/auditoria na mesma transacao Serializable.
RefundReference e referencia unica do estorno manual registrado; nao envia transferencia.
Estorno integral antes de envio/entrega repoe estoque uma vez. Apos envio ou entrega requer retorno fisico e ajuste manual.
Arquivar produto nao remove seus pedidos, recibos, variacoes ou movimentacoes.

model ShopSettings {
  id String @id @default("main")
  mode String @default("INTERNAL") @db.VarChar(20)
  enabled Boolean @default(false)
  externalUrl String @default("") @db.VarChar(1000)
  name String @default("Loja oficial") @db.VarChar(100)
  pickupEnabled Boolean @default(true)
  pickupAddress String @default("") @db.VarChar(500)
  shippingProvider String @default("RULES") @db.VarChar(20)
  originPostalCode String @default("") @db.VarChar(8)
  paymentInstructions String @default("") @db.VarChar(1000)
  supportPhone String @default("") @db.VarChar(11)
  updatedAt DateTime @updatedAt
}

model Product {
  id String @id @default(uuid()) @db.Uuid
  name String @db.VarChar(150)
  description String @default("") @db.VarChar(4000)
  category String @default("Geral") @db.VarChar(80)
  images String[] @default([])
  priceCents Int
  compareAtCents Int?
  costCents Int @default(0)
  weightGrams Int @default(300)
  widthCm Int @default(20)
  heightCm Int @default(5)
  lengthCm Int @default(25)
  active Boolean @default(false)
  featured Boolean @default(false)
  archivedAt DateTime?
  version Int @default(0)
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
  variants ProductVariant[]
  @@index([active, archivedAt, category])
}

model ProductVariant {
  id String @id @default(uuid()) @db.Uuid
  productId String @db.Uuid
  sku String @unique @db.VarChar(80)
  label String @default("Único") @db.VarChar(100)
  stock Int @default(0)
  lowStock Int @default(5)
  active Boolean @default(true)
  version Int @default(0)
  product Product @relation(fields: [productId], references: [id])
  items ShopOrderItem[]
  movements StockMovement[]
  @@index([productId])
}

model StockMovement {
  id String @id @default(uuid()) @db.Uuid
  variantId String @db.Uuid
  actorId String @db.Uuid
  orderId String? @db.Uuid
  requestKey String @unique @db.VarChar(100)
  kind String @db.VarChar(20)
  quantity Int
  balanceAfter Int
  reason String @db.VarChar(500)
  createdAt DateTime @default(now())
  variant ProductVariant @relation(fields: [variantId], references: [id])
  actor User @relation(fields: [actorId], references: [id])
  order ShopOrder? @relation(fields: [orderId], references: [id])
  @@index([variantId, createdAt])
}

  id String @id @default(uuid()) @db.Uuid

  @@index([expiresAt])

model ShopOrder {
  id String @id @default(uuid()) @db.Uuid
  number Int @unique @default(autoincrement())

  requestKey String @db.VarChar(80)
  requestHash String @db.VarChar(64)

  recipient String @db.VarChar(100)

  address Json
  deliveryMethod String @db.VarChar(20)
  shippingName String @db.VarChar(150)
  shippingService String @db.VarChar(80)
  shippingDays Int
  shippingSnapshot Json @default("{}")
  subtotalCents Int
  shippingCents Int
  totalCents Int
  expiresAt DateTime
  inventoryConsumedAt DateTime?
  stockRestoredAt DateTime?
  shippedAt DateTime?
  deliveredAt DateTime?
  trackingCode String @default("") @db.VarChar(100)
  trackingUrl String @default("") @db.VarChar(1000)
  notes String @default("") @db.VarChar(1000)
  version Int @default(0)
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
  user User @relation(fields: [userId], references: [id])
  items ShopOrderItem[]
  payment ShopPayment?
  paymentIntent GatewayIntent?
  movements StockMovement[]
  @@unique([userId, requestKey])
  @@index([status, expiresAt])
  @@index([userId, createdAt])
}

model ShopOrderItem {
  id String @id @default(uuid()) @db.Uuid
  orderId String @db.Uuid
  variantId String @db.Uuid
  productName String @db.VarChar(150)
  variantLabel String @db.VarChar(100)
  sku String @db.VarChar(80)
  image String @default("")
  quantity Int
  unitPriceCents Int
  costCents Int
  order ShopOrder @relation(fields: [orderId], references: [id])
  variant ProductVariant @relation(fields: [variantId], references: [id])
  @@unique([orderId, variantId])
  @@index([variantId])
}

model ShopPayment {
  id String @id @default(uuid()) @db.Uuid
  orderId String @unique @db.Uuid
  method TicketPaymentMethod
  amountCents Int
  reference String @unique @db.VarChar(100)
  refundedCents Int @default(0)
  refundReference String? @unique @db.VarChar(100)
  confirmedById String? @db.Uuid
  confirmedAt DateTime @default(now())
  order ShopOrder @relation(fields: [orderId], references: [id])
  confirmedBy User? @relation(fields: [confirmedById], references: [id])
}
